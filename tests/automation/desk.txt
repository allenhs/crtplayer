# Desk mode (3D set on the desktop). Frame is paused and the preset has no animated
# noise, so every capture is deterministic.
open @M@/sd_4x3_h264.mp4
waitstate playing 15000
seek 5
wait 800
pause
wait 500
preset Clean Broadcast Monitor
# Reference: the regular player in fullscreen
fullscreen on
wait 1500
screenshot filtered @O@/desk_ref_fullscreen.png
fullscreen off
wait 800
desk on
wait 1500
expect deskMode true
expect mainWindowVisible false
expect deskPhase desk
deskgrab @O@/desk_idle.png
report desk-idle
deskfly in
wait 150
deskgrab @O@/desk_flying.png
deskwait full 8000
wait 400
expect deskPhase full
deskgrab @O@/desk_full.png
report desk-full
deskfly out
deskwait desk 8000
wait 400
deskgrab @O@/desk_back.png
report desk-back
