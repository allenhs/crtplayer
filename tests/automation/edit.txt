# (the cuts are written next to them); edit-ro is a read-only folder.
open @O@/edit-media/sd_4x3_h264.mp4

pause
preset Clean Broadcast Monitor
# Keyframe stepping (Shift+→ / Shift+←): the file has a keyframe every 2 s
seek 3.3
wait 800
keyframe next
wait 1200
report kf-next
keyframe prev
wait 1200
report kf-prev1
keyframe prev
wait 1200
report kf-prev2
# A = 3.3 s, B = 7.7 s: the cut starts at the keyframe at 2.0 s
loop 3300 7700
cut
cut
report after-cuts
